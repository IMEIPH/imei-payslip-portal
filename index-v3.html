// Submit Registration Request async function submitRegistration() { const rawEmpId = document.getElementById("regEmpId").value.trim(); // 1. Remove dashes, spaces, special chars & convert to uppercase // e.g., "IS-058" -> "IS058" | "18054-23" -> "1805423" const empId = rawEmpId.replace(/[^a-zA-Z0-9]/g, '').toUpperCase(); // Keep PIN strictly as a string (never parse as a number) const pin = document.getElementById("regPin").value.trim(); const pinConfirm = document.getElementById("regPinConfirm").value.trim(); const statusDiv = document.getElementById("regStatus"); statusDiv.innerHTML = ""; // 2. Check for empty inputs if (!empId || !pin || !pinConfirm) { statusDiv.innerHTML = "PLEASE FILL IN ALL FIELDS."; return; } // 3. Regex check: Exactly 6 digits without converting to Number (preserves '0' at start/end) const pinRegex = /^\d{6}$/; if (!pinRegex.test(pin)) { statusDiv.innerHTML = "PIN MUST BE EXACTLY 6 DIGITS (NUMBERS ONLY)."; return; } // 4. Validate matching PINs if (pin !== pinConfirm) { statusDiv.innerHTML = "PINS DO NOT MATCH. PLEASE TRY AGAIN."; return; } try { statusDiv.innerHTML = "Checking ID..."; // 5. Check if normalized ID already exists in active employees OR pending requests const empDoc = await db.collection("employees").doc(empId).get(); const reqDoc = await db.collection("registration_requests").doc(empId).get(); if (empDoc.exists || reqDoc.exists) { alert("ID already exist"); statusDiv.innerHTML = "ID ALREADY EXIST."; return; } // 6. Save as String to Firebase (Ensures Firestore preserves "012340") await db.collection("registration_requests").doc(empId).set({ empId: String(empId), pin: String(pin), status: "PENDING", requestedAt: firebase.firestore.FieldValue.serverTimestamp() }); statusDiv.innerHTML = "REGISTRATION SUBMITTED! WAITING FOR ADMIN APPROVAL."; setTimeout(() => { closeRegisterModal(); }, 2000); } catch (err) { console.error(err); statusDiv.innerHTML = `ERROR: ${err.message}`; } }
